Benchmark SQL Queries
OLAP aggregation & CTE execution against views
ANALYTICAL SQL 
WITH RouteMetrics AS (
  SELECT 
    r.route_name,
    r.distance_km,
    COALESCE(SUM(pf.fare_amount), 0.0) AS total_revenue
  FROM routes r
  LEFT JOIN trips t ON r.route_id = t.route_id
  LEFT JOIN passenger_fares pf ON t.trip_id = pf.trip_id
  GROUP BY r.route_id
)
SELECT 
  route_name,
  ROUND(total_revenue / distance_km, 2) AS rev_per_km
FROM RouteMetrics
ORDER BY rev_per_km DESC;
